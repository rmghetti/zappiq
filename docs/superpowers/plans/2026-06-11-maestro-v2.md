```typescript
/**
 * Maestro v2 — versionamento de fluxos (spec 2026-06-11).
 * Snapshot imutável a cada publish/refresh/restore; restore cria NOVA versão.
 */
import { prisma } from '@zappiq/database';

export type FlowVersionSource = 'publish' | 'refresh' | 'restore';

interface FlowLike {
  id: string; organizationId: string; name: string;
  nodes: unknown; edges: unknown; triggerType: unknown; triggerConfig: unknown;
}

/** Parte pura: monta os dados do snapshot (testável sem DB). */
export function buildVersionSnapshot(
  flow: FlowLike, version: number, source: FlowVersionSource, createdById: string | null,
) {
  return {
    flowId: flow.id,
    organizationId: flow.organizationId,
    version,
    name: flow.name,
    nodes: (flow.nodes as any) ?? [],
    edges: (flow.edges as any) ?? [],
    triggerType: flow.triggerType as any,
    triggerConfig: (flow.triggerConfig as any) ?? null,
    source,
    createdById,
  };
}

/**
 * Cria snapshot da PRÓXIMA versão do fluxo em transação (atômico contra
 * publicações simultâneas) e devolve o número da versão criada.
 * O chamador é responsável por atualizar Flow.version para o mesmo número.
 */
export async function snapshotFlowVersion(
  flowId: string, organizationId: string, source: FlowVersionSource, createdById: string | null,
): Promise<number> {
  return prisma.$transaction(async (tx) => {
    const flow = await tx.flow.findFirstOrThrow({ where: { id: flowId, organizationId } });
    const last = await tx.flowVersion.findFirst({
      where: { flowId }, orderBy: { version: 'desc' }, select: { version: true },
    });
    const next = Math.max(flow.version, last?.version ?? 0) + 1;
    await tx.flowVersion.create({ data: buildVersionSnapshot(flow as any, next, source, createdById) });
    await tx.flow.update({ where: { id: flowId }, data: { version: next } });
    return next;
  });
}
```

```typescript
/**
 * Maestro v2 — flowRouter (PURO, spec 2026-06-11 Frente 3).
 * Decide QUAL fluxo ativo inicia para uma mensagem inbound, quando a conversa
 * não está mid-flow. Ranking: keyword exata > keyword substring > FIRST_CONTACT
 * (só conversa nova). Desempate: priority desc → updatedAt desc. Determinístico.
 */
export interface RoutableFlow {
  id: string;
  triggerType: string;
  triggerConfig: unknown;
  priority: number;
  updatedAt: Date;
}

export interface RouteContext {
  /** true quando a conversa já concluiu um fluxo (marcador ENDED). */
  conversationEnded: boolean;
}

type MatchRank = 3 | 2 | 1; // 3=keyword exata, 2=keyword substring, 1=first_contact

function keywordsOf(triggerConfig: unknown): string[] {
  const cfg = (triggerConfig as Record<string, any>) || {};
  if (Array.isArray(cfg.keywords)) return cfg.keywords.filter(Boolean).map(String);
  if (typeof cfg.keyword === 'string' && cfg.keyword) return [cfg.keyword];
  return [];
}

function rankFlow(flow: RoutableFlow, message: string, ctx: RouteContext): MatchRank | null {
  const m = (message || '').trim().toLowerCase();
  if (flow.triggerType === 'KEYWORD') {
    const kws = keywordsOf(flow.triggerConfig).map((k) => k.toLowerCase());
    if (kws.length === 0) return null;
    if (kws.some((k) => m === k)) return 3;
    if (kws.some((k) => k && m.includes(k))) return 2;
    return null;
  }
  if (flow.triggerType === 'FIRST_CONTACT') {
    return ctx.conversationEnded ? null : 1;
  }
  // SCHEDULE/MANUAL/EVENT/CUSTOM/CART_ABANDONED/TIMEOUT_*: fail-closed no inbound.
  return null;
}

export function pickFlowForMessage(
  flows: RoutableFlow[], message: string, ctx: RouteContext,
): RoutableFlow | null {
  let best: { flow: RoutableFlow; rank: MatchRank } | null = null;
  for (const flow of flows) {
    const rank = rankFlow(flow, message, ctx);
    if (rank === null) continue;
    if (
      !best ||
      rank > best.rank ||
      (rank === best.rank && flow.priority > best.flow.priority) ||
      (rank === best.rank && flow.priority === best.flow.priority &&
        flow.updatedAt.getTime() > best.flow.updatedAt.getTime())
    ) {
      best = { flow, rank };
    }
  }
  return best?.flow ?? null;
}
```

```typescript
 * FlowTimer (Postgres) é a fonte de verdade; o job atrasado BullMQ é o

import { logger } from '../utils/logger.js';
import { cache } from './cloud/index.js';

  password: redisUrl.password || undefined,

// ── Puro: cálculo do horário de disparo ─────────────────────────────────────
export function computeRunAt(schedule: FlowSchedule, now: Date): Date {
  if (schedule.runAt) {
    const t = new Date(schedule.runAt);
    if (!Number.isNaN(t.getTime()) && t.getTime() > now.getTime()) return t;
    return new Date(now.getTime() + 60_000); // passado/inválido → +1min
  }
  const mins = schedule.delayMinutes ?? 1;
  return new Date(now.getTime() + mins * 60_000);
}

// ── Puro: validação no momento do disparo (fail-closed) ─────────────────────
export interface TimerFireContext {
  timer: { kind: string; flowVersion: number; status: string };
  flow: { isActive: boolean; version: number } | null;
  aiPaused: boolean;
  lastInboundAt: Date | null;
  timerCreatedAt: Date;
  now: Date;
}
export type TimerFireVerdict =
  | { ok: true }
  | { ok: false; status: 'cancelled' | 'expired'; reason: string };

const META_WINDOW_MS = 24 * 60 * 60 * 1000;

export function validateTimerFire(ctx: TimerFireContext): TimerFireVerdict {
  if (ctx.timer.status !== 'pending') return { ok: false, status: 'cancelled', reason: 'not_pending' };
  if (!ctx.flow || !ctx.flow.isActive) return { ok: false, status: 'cancelled', reason: 'flow_inactive' };
  if (ctx.flow.version !== ctx.timer.flowVersion) return { ok: false, status: 'cancelled', reason: 'flow_version_changed' };
  if (ctx.aiPaused) return { ok: false, status: 'cancelled', reason: 'human_takeover' };
  if (ctx.timer.kind === 'wait' && ctx.lastInboundAt && ctx.lastInboundAt.getTime() > ctx.timerCreatedAt.getTime()) {
    return { ok: false, status: 'cancelled', reason: 'customer_replied' };
  }
  if (!ctx.lastInboundAt || ctx.now.getTime() - ctx.lastInboundAt.getTime() > META_WINDOW_MS) {
    return { ok: false, status: 'expired', reason: 'meta_24h_window' };
  }
  return { ok: true };
}

// ── IO: agendar (chamado pelo flowRuntime) ──────────────────────────────────
export async function scheduleFlowTimer(input: {
  organizationId: string; conversationId: string; flowId: string; flowVersion: number;
  schedule: FlowSchedule; state: FlowState;
}): Promise<void> {
  const { organizationId, conversationId, flowId, flowVersion, schedule, state } = input;
  if (!schedule.resumeNodeId) return; // nada pra retomar
  const runAt = computeRunAt(schedule, new Date());
  const timer = await prisma.flowTimer.create({
    data: {
      organizationId, conversationId, flowId, flowVersion,
      kind: schedule.kind, resumeNodeId: schedule.resumeNodeId,
      stateSnapshot: { cursor: schedule.resumeNodeId, vars: state.vars ?? {} } as any,
      runAt,
    },
  });
  const job = await flowTimerQueue.add(
    'fire', { timerId: timer.id },
    { delay: Math.max(0, runAt.getTime() - Date.now()), jobId: `flow-timer:${timer.id}` },
  );
  await prisma.flowTimer.update({ where: { id: timer.id }, data: { jobId: String(job.id) } });
  logger.info('[Maestro] timer agendado', { organizationId, conversationId, flowId, kind: schedule.kind, runAt });
}

// ── IO: cancelar waits pendentes (cliente respondeu) ────────────────────────
export async function cancelPendingWaitTimers(organizationId: string, conversationId: string): Promise<void> {
  const pendings = await prisma.flowTimer.findMany({
    where: { organizationId, conversationId, status: 'pending', kind: 'wait' },
    select: { id: true, jobId: true },
  });
  for (const t of pendings) {
    await prisma.flowTimer.update({ where: { id: t.id }, data: { status: 'cancelled', statusReason: 'customer_replied' } });
    if (t.jobId) {
      try { const j = await flowTimerQueue.getJob(t.jobId); await j?.remove(); } catch { /* job já rodou/expirou */ }
    }
  }
}

let flowTimerWorker: Worker | undefined;

export function initFlowTimerWorker(): void {
  flowTimerWorker = new Worker(
    'flow-timer',
    async (job: Job) => {
      const { timerId } = job.data as { timerId: string };
      const timer = await prisma.flowTimer.findUnique({ where: { id: timerId } });
      if (!timer) return;

      const [flow, conversation, lastInbound] = await Promise.all([
        prisma.flow.findUnique({ where: { id: timer.flowId }, select: { isActive: true, version: true, nodes: true, edges: true } }),
        prisma.conversation.findUnique({ where: { id: timer.conversationId }, select: { aiPaused: true, contactId: true } }),
        prisma.message.findFirst({
          where: { conversationId: timer.conversationId, direction: 'INBOUND' },
          orderBy: { createdAt: 'desc' }, select: { createdAt: true },
        }),
      ]);

      const verdict = validateTimerFire({
        timer: { kind: timer.kind, flowVersion: timer.flowVersion, status: timer.status },
        flow: flow ? { isActive: flow.isActive, version: flow.version } : null,
        aiPaused: Boolean((conversation as any)?.aiPaused),
        lastInboundAt: lastInbound?.createdAt ?? null,
        timerCreatedAt: timer.createdAt,
        now: new Date(),
      });
      if (!verdict.ok) {
        await prisma.flowTimer.update({ where: { id: timer.id }, data: { status: verdict.status, statusReason: verdict.reason } });
        logger.info('[Maestro] timer não disparou', { timerId, reason: verdict.reason });
        return;
      }

      const graph: FlowGraph = { nodes: (flow!.nodes as any) ?? [], edges: (flow!.edges as any) ?? [] };
      const snapshot = (timer.stateSnapshot as any) as FlowState;
      const result = resolveFlowStep(graph, snapshot, '', { hasIncomingMessage: false });

      await executeFlowEffects({
        organizationId: timer.organizationId,
        conversationId: timer.conversationId,
        contactId: (conversation as any)?.contactId ?? null,
        effects: result.effects.filter((e) => e.kind !== 'goto_flow'),
      });

      // Persiste o cursor pós-retomada no MESMO formato do flowRuntime.
      const stateKey = `flow_state:${timer.organizationId}:${timer.conversationId}`;
      const FLOW_STATE_TTL = 60 * 60 * 24 * 7;
      const nextCursor = result.state?.cursor ?? null;
      if (result.next === 'end' || nextCursor === null) {
        await cache.set(stateKey, JSON.stringify({ cursor: '__ended__', vars: result.state?.vars || {} }), FLOW_STATE_TTL);
      } else {
        await cache.set(stateKey, JSON.stringify({ ...result.state, flowId: timer.flowId }), FLOW_STATE_TTL);
      }

      await prisma.flowTimer.update({ where: { id: timer.id }, data: { status: 'fired', firedAt: new Date() } });
      logger.info('[Maestro] timer disparado e fluxo retomado', { timerId, next: result.next, effects: result.effects.map((e) => e.kind) });
    },
    { connection },
  );
  flowTimerWorker.on('failed', (job, err) => {
    logger.error('[Maestro] flow-timer job falhou', { jobId: job?.id, err: String(err) });
  });
  logger.info('[Queues] flow-timer worker iniciado');
}
```

```typescript
/**
 * Maestro v2 — patch de conteúdo com ESTRUTURA TRAVADA (spec Frente 2).
 * Puro: extrai/aplica/difere conteúdo textual de nós sem tocar ids/edges/posições.
 */
export interface ContentField { nodeId: string; field: 'text' | 'tag' | 'prompt'; value: string }
export interface ContentDiff { nodeId: string; field: string; before: string; after: string }

const FIELD_BY_TYPE: Record<string, 'text' | 'tag' | 'prompt'> = {
  message: 'text', tag: 'tag', ai: 'prompt',
};

export function extractEditableContent(nodes: any[]): ContentField[] {
  const out: ContentField[] = [];
  for (const n of nodes ?? []) {
    const field = FIELD_BY_TYPE[n?.type];
    if (!field) continue;
    out.push({ nodeId: String(n.id), field, value: String(n?.data?.[field] ?? '') });
  }
  return out;
}

export function applyContentPatch(nodes: any[], patch: ContentField[]): any[] {
  const byNode = new Map(patch.map((p) => [`${p.nodeId}:${p.field}`, p.value]));
  return (nodes ?? []).map((n) => {
    const field = FIELD_BY_TYPE[n?.type];
    if (!field) return n;
    const key = `${n.id}:${field}`;
    if (!byNode.has(key)) return n;
    return { ...n, data: { ...n.data, [field]: byNode.get(key) } };
  });
}

export function diffContent(nodes: any[], patch: ContentField[]): ContentDiff[] {
  const current = new Map(extractEditableContent(nodes).map((c) => [`${c.nodeId}:${c.field}`, c.value]));
  const diffs: ContentDiff[] = [];
  for (const p of patch) {
    const key = `${p.nodeId}:${p.field}`;
    if (!current.has(key)) continue;
    const before = current.get(key)!;
    if (before !== p.value) diffs.push({ nodeId: p.nodeId, field: p.field, before, after: p.value });
  }
  return diffs;
}
```

```typescript
export function sameStructure(current: any[], proposed: any[]): boolean {
```
